     4	function tool(name: string, description: string, extra: Record<string, unknown> = {}): unknown {
     5	  return {
     6	    type: "function",
     7	    function: {
     8	      name,
     9	      description,
    10	      parameters: {
    11	        $schema: "https://json-schema.org/draft/2020-12/schema",
    12	        type: "object",
    13	        properties: {
    14	          accountId: {
    15	            type: "string",
    16	            title: "accountId",
    17	            description: "The customer account identifier as shown in the admin console.",
    18	          },
    19	          limit: { type: "integer", title: "Maximum results", description: "Between 1 and 100." },
    20	        },
    21	        required: ["accountId"],
    22	        ...extra,
    23	      },
    24	    },
    25	  };
    26	}

   184	/** Resolve a local "#/…" JSON pointer against a document. */
   185	function resolvePointer(document: unknown, reference: string): unknown {
   186	  let node: unknown = document;
   187	  for (const raw of reference.slice(1).split("/").filter((part) => part !== "")) {
   188	    const key = decodeURIComponent(raw).replace(/~1/g, "/").replace(/~0/g, "~");
   189	    if (typeof node !== "object" || node === null || !(key in node)) return undefined;
   190	    node = (node as Record<string, unknown>)[key];
   191	  }
   192	  return node;
   193	}

   195	/** Every "$ref" string in a schema. */
   196	function referencesIn(node: unknown, found: string[] = []): string[] {
   197	  if (Array.isArray(node)) {
   198	    for (const item of node) referencesIn(item, found);
   199	  } else if (typeof node === "object" && node !== null) {
   200	    for (const [key, value] of Object.entries(node)) {
   201	      if (key === "$ref" && typeof value === "string") found.push(value);
   202	      else referencesIn(value, found);
   203	    }
   204	  }
   205	  return found;
   206	}
