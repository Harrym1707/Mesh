+/** True when anything in `node` uses a reference keyword. Conservative: it also
+ *  counts one inside data, which only ever keeps more. */
+function usesReferences(node: unknown): boolean {
+  if (Array.isArray(node)) return node.some(usesReferences);
+  if (typeof node !== "object" || node === null) return false;

+    if (usesReferences(value)) return true;
+  }
